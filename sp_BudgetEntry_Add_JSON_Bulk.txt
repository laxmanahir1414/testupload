ALTER PROCEDURE [dbo].[sp_BudgetEntry_Bulk_Save_JSON]
(
   --Declare 
		@EntryType INT,                -- 1 Budget / 2 Estimation / 3 Actual
		@QuarterCode INT,              -- 1 / 2 / 3 / 4
		@FY_Code INT,
		@RoleName VARCHAR(200),
		@Created_By INT,
		@Payload NVARCHAR(MAX),

		@ExecStatus BIT OUTPUT,
		@ResultMessage NVARCHAR(1000) OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    SET @ExecStatus = 0;
    SET @ResultMessage = NULL;

    DECLARE
        @RoleCode INT,
        @FY_Status INT,
		@IsCurrent BIT,
        @QuarterStatus INT,
        @PreviousQuarterStatus INT;

    BEGIN TRY

				 -- ==============================================================
				-- 1. Basic validation
				-- ==============================================================
				IF ISJSON(@Payload) <> 1
					THROW 73000, 'Invalid JSON payload.', 1;

				IF @EntryType NOT IN (1, 2, 3)
					THROW 73001, 'Invalid Entry Type. 1=Budget, 2=Estimation, 3=Actual.', 1;

				IF @QuarterCode NOT IN (1, 2, 3, 4)
					THROW 73002, 'Invalid Quarter Code. Use 1, 2, 3 or 4.', 1;

				IF ISNULL(@FY_Code, 0) <= 0
					THROW 73003, 'Financial Year is required.', 1;

				IF ISNULL(@Created_By, 0) <= 0
					THROW 73004, 'Created By user is required.', 1;

				-- ==============================================================
				-- 2. Role validation
				-- ==============================================================
				SELECT @RoleCode = role_code
				FROM access.roles
				WHERE role_name = @RoleName AND is_active = 1;

				IF @RoleCode IS NULL
					THROW 73005, 'Invalid or inactive role.', 1;

				-- ==============================================================
				-- 3. Financial Year validation
				-- ==============================================================
				SELECT @FY_Status = Status, @IsCurrent = Is_Current
				FROM masters.FinancialYears
				WHERE FY_Code = @FY_Code AND is_active = 1;

				IF @FY_Status IS NULL
					THROW 73006, 'Invalid or inactive Financial Year.', 1;

				IF @FY_Status = 4
					THROW 73007, 'Financial Year is Closed. No add/update allowed.', 1;

				IF @IsCurrent <> 1
					THROW 73008, 'Financial Year must be marked as Current.', 1;

				-- ==============================================================
				-- 4. Quarter validation
				-- ==============================================================
				SELECT @QuarterStatus = Status
				FROM masters.Quarters
				WHERE FY_Code = @FY_Code AND Quarter_Code = @QuarterCode AND is_active = 1;

				IF @QuarterStatus IS NULL
					THROW 73009, 'Quarter is invalid or not configured.', 1;

				IF @QuarterStatus = 4
					THROW 73010, 'Quarter is Closed. No add/update allowed.', 1;

	

		   -- ================================================================
			   --5. PREVIOUS QUARTER CHECK

			   --The uploaded rule says that when opening the next quarter, the
			   --previous quarter status must be checked for Estimation Frozen or
			   --Closed.

			   --This procedure does not change Quarter master status. Therefore
			   --it validates that a previous-quarter record exists when processing
			   --Q2-Q4, and rejects an invalid status value.
			--   ================================================================ 

			IF @QuarterCode > 1
				BEGIN
					SELECT @PreviousQuarterStatus = Status
					FROM masters.Quarters
					WHERE FY_Code = @FY_Code
					  AND quarter_code = @QuarterCode - 1
					  AND is_active = 1;

					IF @PreviousQuarterStatus IS NULL
						THROW 73014,
							  'Previous quarter is not configured for the selected Financial Year.',
							  1;

					IF @PreviousQuarterStatus = 2--NOT IN (2, 3, 4)
						THROW 73015,
							  'Previous quarter has an invalid status. It must be Frozen, or Closed.',
							  1;
				END

        -- ================================================================
           --6. LOAD EXCEL JSON
        --   ================================================================ 

        CREATE TABLE #Upload
        (
            RowNumber INT NOT NULL,
            SrNo INT NULL,
            BudgetCode VARCHAR(30) NULL,

            ProjectName NVARCHAR(200) NULL,
            MandatedCompany NVARCHAR(200) NULL,
            Industry NVARCHAR(200) NULL,
            Product NVARCHAR(200) NULL,
            IssueType NVARCHAR(200) NULL,

            IndicativeDealSize DECIMAL(18,2) NULL,
            EstimatedFeePool DECIMAL(18,2) NULL,
            TotalFees DECIMAL(18,2) NULL,
            NoOfBRLMs INT NULL,
            JMFKeepRate DECIMAL(18,2) NULL,
            JMFGrossFee DECIMAL(18,2) NULL,
            DealProbability DECIMAL(18,2) NULL,
            JMProbability DECIMAL(18,2) NULL,

            ExpectedClosure NVARCHAR(20) NULL,
            Q1 DECIMAL(18,2) NULL,
            Q2 DECIMAL(18,2) NULL,
            Q3 DECIMAL(18,2) NULL,
            Q4 DECIMAL(18,2) NULL,
            Total DECIMAL(18,2) NULL,

            EntryQuarter NVARCHAR(20) NULL,
            Amount DECIMAL(18,2) NULL,
            IsManualCalculation VARCHAR(10) NULL,

            RowJson NVARCHAR(MAX) NOT NULL
        );

        INSERT INTO #Upload
        (
            RowNumber, SrNo, BudgetCode,
            ProjectName, MandatedCompany, Industry, Product, IssueType,
            IndicativeDealSize, EstimatedFeePool, TotalFees, NoOfBRLMs,
            JMFKeepRate, JMFGrossFee, DealProbability, JMProbability,
            ExpectedClosure, Q1, Q2, Q3, Q4, Total,
            EntryQuarter, Amount, IsManualCalculation, RowJson
        )
        SELECT
            TRY_CONVERT(INT, [key]) + 1,
            TRY_CONVERT(INT, JSON_VALUE(value, '$."Sr No"')),
            NULLIF(LTRIM(RTRIM(JSON_VALUE(value, '$."Budget Code"'))), ''),

            JSON_VALUE(value, '$."Project Name"'),
            JSON_VALUE(value, '$."Mandated Company"'),
            JSON_VALUE(value, '$."Industry"'),
            JSON_VALUE(value, '$."Product"'),
            JSON_VALUE(value, '$."Issue Type"'),

            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Indicative Deal Size (Rs Cr)"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Estimated Fee Pool (%)"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Total Fees (Rs Cr)"')),
            TRY_CONVERT(INT, JSON_VALUE(value, '$."No. of BRLMs"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."JMF Keep Rate (%)"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."JMF Gross Fee (Rs Cr)"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Deal Probability (%)"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."JM Probability (%)"')),

            JSON_VALUE(value, '$."Expected Closure"'),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Q1"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Q2"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Q3"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Q4"')),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Total"')),

            JSON_VALUE(value, '$."Quarter"'),
            TRY_CONVERT(DECIMAL(18,2), JSON_VALUE(value, '$."Amount"')),
            JSON_VALUE(value, '$."Is Manual Calculation"'),
            value
			FROM OPENJSON(@Payload);


			IF NOT EXISTS (SELECT 1 FROM #Upload)
				THROW 73016, 'No budget records found in uploaded file.', 1;


			/* ================================================================
			   7. BATCH VALIDATION - NO DATA IS SAVED IF ANY ROW FAILS
			   ================================================================ */

			CREATE TABLE #ValidationErrors
			(
				RowNumber INT NOT NULL,
				BudgetCode VARCHAR(30) NULL,
				ErrorMessage NVARCHAR(1000) NOT NULL
			);

			-- ==============================================================
			-- EntryType-specific rules (batch validation)
			-- ==============================================================
			-- Budget Add
			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Budget Code already exists. Must be unique for Add.'
			FROM #Upload u
			WHERE @EntryType = 1
			AND EXISTS (SELECT 1 FROM dbo.BudgetEntryGroupMaster gm WHERE gm.Budget_Code = u.BudgetCode);

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Quarter must be Open for Budget Add.'
			FROM #Upload u
			WHERE @EntryType = 1 AND @QuarterStatus <> 2;

			-- Budget Edit
			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Budget Code does not exist. Cannot edit.'
			FROM #Upload u
			WHERE @EntryType = 1
			AND NOT EXISTS (SELECT 1 FROM dbo.BudgetEntryGroupMaster gm WHERE gm.Budget_Code = u.BudgetCode);

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Quarter must be Open for Budget Edit.'
			FROM #Upload u
			WHERE @EntryType = 1 AND @QuarterStatus <> 2;

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Actual already exists. Budget cannot be edited.'
			FROM #Upload u
			WHERE @EntryType = 1
			AND EXISTS (
			SELECT 1
			FROM dbo.BudgetEntry b
			INNER JOIN dbo.BudgetEntryGroup g ON b.BudgetEntryGroup_Syscode = g.BudgetEntryGroup_Syscode
			INNER JOIN dbo.BudgetEntryGroupMaster gm ON g.BudgetEGMaster_syscode = gm.BudgetEGMaster_syscode
			WHERE gm.Budget_Code = u.BudgetCode
			AND g.FY_Code = @FY_Code
			AND b.Quarter_Code = @QuarterCode
			AND g.BET_Code = 3
			);

			-- Estimation Entry
			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Budget Code does not exist. Estimation requires existing Budget.'
			FROM #Upload u
			WHERE @EntryType = 2
			AND NOT EXISTS (SELECT 1 FROM dbo.BudgetEntryGroupMaster gm WHERE gm.Budget_Code = u.BudgetCode);

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Quarter must be Open for Estimation.'
			FROM #Upload u
			WHERE @EntryType = 2 AND @QuarterStatus <> 2;

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Actual exists. Estimation cannot be edited.'
			FROM #Upload u
			WHERE @EntryType = 2
			AND EXISTS (
			SELECT 1
			FROM dbo.BudgetEntry b
			INNER JOIN dbo.BudgetEntryGroup g ON b.BudgetEntryGroup_Syscode = g.BudgetEntryGroup_Syscode
			INNER JOIN dbo.BudgetEntryGroupMaster gm ON g.BudgetEGMaster_syscode = gm.BudgetEGMaster_syscode
			WHERE gm.Budget_Code = u.BudgetCode
			AND g.FY_Code = @FY_Code
			AND b.Quarter_Code = @QuarterCode
			AND g.BET_Code = 3
			);

			-- Actual Entry
			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Budget Code does not exist. Actual requires existing Budget.'
			FROM #Upload u
			WHERE @EntryType = 3
			AND NOT EXISTS (SELECT 1 FROM dbo.BudgetEntryGroupMaster gm WHERE gm.Budget_Code = u.BudgetCode);

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Quarter must be Open for Actual.'
			FROM #Upload u
			WHERE @EntryType = 3 AND @QuarterStatus <> 2;

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode, 'Estimation must exist before Actual entry.'
			FROM #Upload u
			WHERE @EntryType = 3
			AND NOT EXISTS (
			SELECT 1
			FROM dbo.BudgetEntry b
			INNER JOIN dbo.BudgetEntryGroup g ON b.BudgetEntryGroup_Syscode = g.BudgetEntryGroup_Syscode
			INNER JOIN dbo.BudgetEntryGroupMaster gm ON g.BudgetEGMaster_syscode = gm.BudgetEGMaster_syscode
			WHERE gm.Budget_Code = u.BudgetCode
			AND g.FY_Code = @FY_Code
			AND b.Quarter_Code = @QuarterCode
			AND g.BET_Code = 2
			);


			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT RowNumber, BudgetCode, 'Budget Code is required.'
			FROM #Upload
			WHERE BudgetCode IS NULL;

			/*
			For one bulk call, the same Budget Code represents the same
			EntryType + Quarter. Two rows would create an ambiguous version chain,
			so reject duplicates.
			*/
			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT u.RowNumber, u.BudgetCode,
				   'Duplicate Budget Code found in this Excel batch.'
			FROM #Upload u
			INNER JOIN
			(
				SELECT BudgetCode
				FROM #Upload
				WHERE BudgetCode IS NOT NULL
				GROUP BY BudgetCode
				HAVING COUNT(*) > 1
			) d
				ON d.BudgetCode = u.BudgetCode;

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT RowNumber, BudgetCode,
				   'Project Name or Mandated Company is required.'
			FROM #Upload
			WHERE NULLIF(LTRIM(RTRIM(ProjectName)), '') IS NULL
			  AND NULLIF(LTRIM(RTRIM(MandatedCompany)), '') IS NULL;

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT RowNumber, BudgetCode,
				   'Deal Probability (%) must be between 0 and 100.'
			FROM #Upload
			WHERE DealProbability IS NOT NULL
			  AND (DealProbability < 0 OR DealProbability > 100);

			INSERT INTO #ValidationErrors (RowNumber, BudgetCode, ErrorMessage)
			SELECT RowNumber, BudgetCode,
				   'JM Probability (%) must be between 0 and 100.'
			FROM #Upload
			WHERE JMProbability IS NOT NULL
			  AND (JMProbability < 0 OR JMProbability > 100);

			IF EXISTS (SELECT 1 FROM #ValidationErrors)
			BEGIN
				SELECT RowNumber, BudgetCode, ErrorMessage
				FROM #ValidationErrors
				ORDER BY RowNumber, ErrorMessage;

				SET @ExecStatus = 0;
				SET @ResultMessage =
					'Excel validation failed. No Budget, Estimation or Actual records were saved.';

				RETURN;
			END;


        /* ================================================================
           8. SAVE ALL ROWS IN ONE TRANSACTION
           ================================================================ */

        BEGIN TRANSACTION;

        DECLARE
            @RowNumber INT,
            @BudgetCode VARCHAR(30),
            @RowJson NVARCHAR(MAX),

            @BudgetEGMasterSyscode BIGINT,
            @LatestVersionEntrySyscode BIGINT,
            @BudgetBaselineEntrySyscode BIGINT,

            @WorkerPayload NVARCHAR(MAX),
            @WorkerStatus BIT,
            @WorkerMessage NVARCHAR(1000);

        DECLARE BudgetCursor CURSOR LOCAL FAST_FORWARD FOR
        SELECT RowNumber, BudgetCode, RowJson
        FROM #Upload
        ORDER BY RowNumber;

        OPEN BudgetCursor;

        FETCH NEXT FROM BudgetCursor
        INTO @RowNumber, @BudgetCode, @RowJson;

        WHILE @@FETCH_STATUS = 0
        BEGIN
            SET @BudgetEGMasterSyscode = NULL;
            SET @LatestVersionEntrySyscode = NULL;
            SET @BudgetBaselineEntrySyscode = NULL;
            SET @WorkerStatus = 0;
            SET @WorkerMessage = NULL;


            /* ============================================================
               8A. RESOLVE BUDGET CODE ROOT
               ============================================================ */

            SELECT
                @BudgetEGMasterSyscode = BudgetEGMaster_syscode
            FROM dbo.BudgetEntryGroupMaster WITH (UPDLOCK, HOLDLOCK)
            WHERE Budget_Code = @BudgetCode;

            IF @BudgetEGMasterSyscode IS NULL
            BEGIN
                /*
                New Budget Code can only start with Budget.
                Estimation/Actual cannot create an orphan lifecycle.
                */
                IF @EntryType <> 1
                    THROW 73020,
                          'Budget Code does not exist. Estimation or Actual requires an existing Budget.',
                          1;

                INSERT INTO dbo.BudgetEntryGroupMaster
                (
                    Budget_Code,
                    is_active,
                    created_date
                )
                VALUES
                (
                    @BudgetCode,
                    1,
                    GETDATE()
                );

                SET @BudgetEGMasterSyscode =
                    CONVERT(BIGINT, SCOPE_IDENTITY());
            END;


            /* ============================================================
               8B. FIND THE LATEST VERSION FOR THIS EXACT STREAM

               Stream identity:
                 Budget Code root
                 Financial Year
                 Entry Type
                 Quarter

               We do NOT UPDATE this record.
               Its id becomes Supersedes_BEntrySys for the new version.
               ============================================================ */

            SELECT TOP (1)
                @LatestVersionEntrySyscode = be.BudgetEntry_Syscode
            FROM dbo.BudgetEntry be
            INNER JOIN dbo.BudgetEntryGroup beg
                ON beg.BudgetEntryGroup_Syscode =
                   be.BudgetEntryGroup_Syscode
            WHERE beg.BudgetEGMaster_syscode = @BudgetEGMasterSyscode
              AND beg.FY_Code = @FY_Code
              AND beg.BET_Code = @EntryType
              AND be.quarter_code = @QuarterCode
              AND ISNULL(beg.Is_Deleted, 0) = 0
              AND ISNULL(be.Is_Deleted, 0) = 0
            ORDER BY
                be.BudgetEntry_Syscode DESC;


            /* ============================================================
               8C. ESTIMATION CANNOT CREATE A NEW VERSION AFTER ACTUAL
               EXISTS FOR THE SAME BUDGET CODE + FY + QUARTER
               ============================================================ */

            IF @EntryType = 2
            BEGIN
                IF EXISTS
                (
                    SELECT 1
                    FROM dbo.BudgetEntry a
                    INNER JOIN dbo.BudgetEntryGroup ag
                        ON ag.BudgetEntryGroup_Syscode =
                           a.BudgetEntryGroup_Syscode
                    WHERE ag.BudgetEGMaster_syscode = @BudgetEGMasterSyscode
                      AND ag.FY_Code = @FY_Code
                      AND ag.BET_Code = 3
                      AND a.quarter_code = @QuarterCode
                      AND ISNULL(ag.Is_Deleted, 0) = 0
                      AND ISNULL(a.Is_Deleted, 0) = 0
                )
                    THROW 73021,
                          'Estimation cannot be added or versioned because Actual already exists for this Budget Code and Quarter.',
                          1;
            END;


            /* ============================================================
               8D. ACTUAL REQUIRES A BUDGET BASELINE

               Latest Budget version is used as the baseline reference.
               If your existing business rule requires the ORIGINAL first
               budget instead, change DESC to ASC here.
               ============================================================ */

            IF @EntryType = 3
            BEGIN
                SELECT TOP (1)
                    @BudgetBaselineEntrySyscode = be.BudgetEntry_Syscode
                FROM dbo.BudgetEntry be
                INNER JOIN dbo.BudgetEntryGroup beg
                    ON beg.BudgetEntryGroup_Syscode =
                       be.BudgetEntryGroup_Syscode
                WHERE beg.BudgetEGMaster_syscode = @BudgetEGMasterSyscode
                  AND beg.FY_Code = @FY_Code
                  AND beg.BET_Code = 1
                  AND ISNULL(beg.Is_Deleted, 0) = 0
                  AND ISNULL(be.Is_Deleted, 0) = 0
                ORDER BY be.BudgetEntry_Syscode DESC;

                IF @BudgetBaselineEntrySyscode IS NULL
                    THROW 73022,
                          'Actual cannot be added because a Budget baseline does not exist for this Budget Code.',
                          1;
            END;


            /* ============================================================
               8E. BUILD THE EXISTING SINGLE-ROW PROCEDURE PAYLOAD

               IMPORTANT VERSIONING VALUES:

                 budgetentry_syscode
                     = latest version id.
                       The existing procedure uses this for its existing-entry
                       validation (for example pending request checks).

                 Supersedes_BEntrySys
                     = latest version id.
                       The NEW BudgetEntry must point to the previous version.

               On first insert both values are NULL/0.
               ============================================================ */

            SET @WorkerPayload = @RowJson;

            SET @WorkerPayload =
                JSON_MODIFY(@WorkerPayload, '$.userSyscode', @Created_By);

            SET @WorkerPayload =
                JSON_MODIFY(@WorkerPayload, '$.fy_Code', @FY_Code);

            SET @WorkerPayload =
                JSON_MODIFY(@WorkerPayload, '$.roleCode', @RoleCode);

            SET @WorkerPayload =
                JSON_MODIFY(@WorkerPayload, '$.bet_Code', @EntryType);

            SET @WorkerPayload =
                JSON_MODIFY(@WorkerPayload, '$.quarter_code', @QuarterCode);

            SET @WorkerPayload =
                JSON_MODIFY
                (
                    @WorkerPayload,
                    '$.budgetEGMaster_syscode',
                    @BudgetEGMasterSyscode
                );

            SET @WorkerPayload =
                JSON_MODIFY
                (
                    @WorkerPayload,
                    '$.budgetentry_syscode',
                    @LatestVersionEntrySyscode
                );

            SET @WorkerPayload =
                JSON_MODIFY
                (
                    @WorkerPayload,
                    '$.Supersedes_BEntrySys',
                    @LatestVersionEntrySyscode
                );

            SET @WorkerPayload =
                JSON_MODIFY
                (
                    @WorkerPayload,
                    '$.BudgetBaseline_BEntrySys',
                    @BudgetBaselineEntrySyscode
                );


            /* ============================================================
               8F. CALL EXISTING VERSION-CREATING PROCEDURE

               It must INSERT a new BudgetEntryGroup and BudgetEntry.
               It must NOT UPDATE the previous BudgetEntry row.
               ============================================================ */

            EXEC dbo.sp_BudgetEntry_Add_JSON
                @Payload = @WorkerPayload,
                @ExecStatus = @WorkerStatus OUTPUT,
                @ResultMessage = @WorkerMessage OUTPUT;

            IF ISNULL(@WorkerStatus, 0) = 0
                THROW 73023, @WorkerMessage, 1;


            FETCH NEXT FROM BudgetCursor
            INTO @RowNumber, @BudgetCode, @RowJson;
			END;

			CLOSE BudgetCursor;
			DEALLOCATE BudgetCursor;

			COMMIT TRANSACTION;

			SET @ExecStatus = 1;
			SET @ResultMessage =
				'All rows were saved successfully. Existing entries were preserved and new versions were created where applicable.';

				SELECT
					u.RowNumber,
					u.BudgetCode,
					CASE
						WHEN EXISTS
						(
							SELECT 1
							FROM dbo.BudgetEntry be
							INNER JOIN dbo.BudgetEntryGroup beg
								ON beg.BudgetEntryGroup_Syscode =
									be.BudgetEntryGroup_Syscode
							WHERE beg.BudgetEGMaster_syscode =
							(
								SELECT BudgetEGMaster_syscode
								FROM dbo.BudgetEntryGroupMaster
								WHERE Budget_Code = u.BudgetCode
							)
								AND beg.FY_Code = @FY_Code
								AND beg.BET_Code = @EntryType
								AND be.quarter_code = @QuarterCode
								AND be.Supersedes_BEntrySys IS NOT NULL
						)
						THEN 'VERSION_CREATED'
						ELSE 'CREATED'
					END AS SaveStatus
				FROM #Upload u
				ORDER BY u.RowNumber;

    END TRY
    BEGIN CATCH

        IF CURSOR_STATUS('local', 'BudgetCursor') >= -1
        BEGIN
            CLOSE BudgetCursor;
            DEALLOCATE BudgetCursor;
        END;

        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        DECLARE
            @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE(),
            @ErrorNumber INT = ERROR_NUMBER(),
            @ErrorLine INT = ERROR_LINE();

        SET @ExecStatus = 0;
        SET @ResultMessage =
            CONCAT(
                'Bulk save failed. Error ',
                @ErrorNumber,
                ' at line ',
                @ErrorLine,
                ': ',
                @ErrorMessage
            );

        SELECT
            CAST(NULL AS INT) AS RowNumber,
            CAST(NULL AS VARCHAR(30)) AS BudgetCode,
            'FAILED' AS SaveStatus,
            @ResultMessage AS ErrorMessage;
    END CATCH
END;

ALTER PROCEDURE [dbo].[sp_BudgetEntry_Add_JSON]
    @Payload NVARCHAR (MAX), @ExecStatus BIT OUTPUT, @ResultMessage NVARCHAR (500) OUTPUT
AS
BEGIN 
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
    BEGIN TRY
        IF ISJSON(@Payload) = 0
            THROW 71000, 'Invalid JSON payload.', 1;
        DECLARE @UserSyscode AS INT, @FY_Code AS INT, @UserEntityCompCode AS INT, @IndusCGCode AS INT, @Cproject_Code AS INT, 
		@Ccompany_Code AS INT, @Cindustry_code AS INT, @Cproject_Name AS NVARCHAR (200), @Ccompany_Name AS NVARCHAR (200),
		@Cindustry_Name AS NVARCHAR (100), @Product_code AS INT, @ProdIssueType_code AS INT, @IndicativeDealSize AS DECIMAL (18, 2), 
		@EstimatedFeePool AS DECIMAL (5, 2), @TotalFees AS DECIMAL (18, 2), @NoOfBRLMs AS INT, @JMFKeepRate AS DECIMAL (5, 2),
		@JMFGrossFee AS DECIMAL (18, 2), @DealProbability AS DECIMAL (5, 2), @JMProbability AS DECIMAL (5, 2),
		@ExpectedClosure AS INT, @BudgetBaseline_BEntrySys AS BIGINT, @RoleCode AS INT, @BET_Code AS INT,
		@QuarterCode AS INT, @Q1Fees AS DECIMAL (18, 2), @Q2Fees AS DECIMAL (18, 2), @Q3Fees AS DECIMAL (18, 2),
		@Q4Fees AS DECIMAL (18, 2), @TotalEstimate AS DECIMAL (18, 2),@isMandated bit,@budgetEntry_Syscode BIGINT,
        @net_amount AS DECIMAL(18, 2),@fee_amount AS DECIMAL(18, 2),@remarks as nvarchar(200), @gross_amount AS DECIMAL(18, 2),
        @BudgetEGMaster_syscode int ;
        
		SELECT @UserSyscode = UserSyscode,
               @FY_Code = FY_Code,
               @UserEntityCompCode = UserEntityCompCode,
               @IndusCGCode = IndusCGCode,
               @Cproject_Code = Cproject_Code,
               @Ccompany_Code = Ccompany_Code,
               @Cindustry_code = Cindustry_code,
               @Cproject_Name = Cproject_Name,
               @Ccompany_Name = Ccompany_Name,
               @Cindustry_Name = Cindustry_Name,
               @Product_code = Product_code,
               @ProdIssueType_code = ProdIssueType_code,
               @IndicativeDealSize = IndicativeDealSize,
               @EstimatedFeePool = EstimatedFeePool,
               @TotalFees = TotalFees,
               @NoOfBRLMs = NoOfBRLMs,
               @JMFKeepRate = JMFKeepRate,
               @JMFGrossFee = JMFGrossFee,
               @DealProbability = DealProbability,
               @JMProbability = JMProbability,
               @ExpectedClosure = ExpectedClosure,

			   @isMandated=isMandated,

               @net_amount=net_amount,
               @fee_amount=fee_amount,
               @remarks=remarks,

        FROM   OPENJSON (@Payload) WITH (userSyscode INT, fy_Code INT, userEntityCompCode INT,indusCGCode INT,

		budgetEntry_Syscode BIGINT,net_amount DECIMAL (18, 2),fee_amount DECIMAL (18, 2),remarks nvarchar(200),budgetEGMaster_syscode int);

        IF OBJECT_ID('tempdb..#temp_fee_dtls') IS NOT NULL
        BEGIN  
               DROP TABLE #temp_fee_dtls;
        END

        SELECT *
        INTO #temp_fee_dtls
        FROM OPENJSON(@Payload, '$.fee_dtls')
        WITH (
            budget_fee_dtl_syscode INT '$.budget_fee_dtl_syscode',
            BudgetEntry_Syscode INT '$.BudgetEntry_Syscode',
            fee_code INT '$.fee_code',
            fee_type NVARCHAR(200) '$.fee_type',
            Q1fee DECIMAL(18,2) '$.Q1fee',
            Q2fee DECIMAL(18,2) '$.Q2fee',
            Q3fee DECIMAL(18,2) '$.Q3fee',
            Q4fee DECIMAL(18,2) '$.Q4fee'
        );

             IF OBJECT_ID('tempdb..#temp_budget_fee_sharing') IS NOT NULL
            BEGIN  
                DROP TABLE #temp_budget_fee_sharing;
            END

            -- Explicitly create the temp table structure
            CREATE TABLE #temp_budget_fee_sharing (
                budgetFeeSharing_syscode INT NULL,
                BudgetEntry_Syscode INT NULL,
                fee_type_code INT NULL,
                fee_type_name NVARCHAR(200) NULL,
                sharing_amt DECIMAL(18,2) NULL,
                jmfs_amt DECIMAL(18,2) NULL,
                jmfied_amt DECIMAL(18,2) NULL,
                jmflibd_amt DECIMAL(18,2) NULL,
                is_sharing_on BIT NULL
            );

             -- First insert from JSON payload
            INSERT INTO #temp_budget_fee_sharing
            SELECT 
                budgetFeeSharing_syscode,
                BudgetEntry_Syscode,
                fee_type_code,
                fee_type_name,
                sharing_amt,
                jmfs_amt,
                jmfied_amt,
                jmflibd_amt,
                is_sharing_on
            FROM OPENJSON(@Payload, '$.fee_sharing_dtls')
            WITH (
                budgetFeeSharing_syscode INT '$.budgetFeeSharing_syscode',
                BudgetEntry_Syscode INT '$.BudgetEntry_Syscode',
                fee_type_code INT '$.fee_type_code',
                fee_type_name NVARCHAR(200) '$.fee_type_name',
                sharing_amt DECIMAL(18,2) '$.sharing_amt',
                jmfs_amt DECIMAL(18,2) '$.jmfs_amt',
                jmfied_amt DECIMAL(18,2) '$.jmfied_amt',
                jmflibd_amt DECIMAL(18,2) '$.jmflibd_amt',
                is_sharing_on BIT '$.is_sharing_on'
            );

        BEGIN TRANSACTION;
       
        DECLARE @AllowControllerAddBudget AS BIT = (SELECT PolicyValue
                                                    FROM   masters.PolicyFlags
                                                    WHERE  PolicyCode = 'AllowControllerAddBudget'),
		@AllowControllerAddEstimated AS BIT = (SELECT PolicyValue FROM masters.PolicyFlags
											WHERE  PolicyCode = 'AllowControllerAddEstimated'), 
		@RequireMakerResp AS BIT = (SELECT PolicyValue  FROM masters.PolicyFlags
											WHERE  PolicyCode = 'RequireMakerResponsibilityForUpdate'), 
		@EnforceFYOpen AS BIT = (SELECT PolicyValue FROM  masters.PolicyFlags
											WHERE  PolicyCode = 'EnforceFYOpenForEdits'),
		@EnforceQtrBudgetUser AS BIT = (SELECT PolicyValue FROM   masters.PolicyFlags 
										WHERE  PolicyCode = 'EnforceQuarterStatusForBudgetUser'), 
		@EnforceQtrController AS BIT = (SELECT PolicyValue  FROM   masters.PolicyFlags
										WHERE  PolicyCode = 'EnforceQuarterStatusForController');

        IF NOT EXISTS (SELECT 1
                       FROM   access.roles
                       WHERE  role_code = @RoleCode
                              AND is_active = 1)
            THROW 71001, 'Invalid or inactive role.', 1;

            if exists(select 1 from BudgetEntry be inner join BudgetEntryGroup bg on 
            be.BudgetEntryGroup_Syscode=bg.BudgetEntryGroup_Syscode where be.BudgetEntry_Syscode=@budgetEntry_Syscode and be.Status=0)
                throw  71001, 'Change request already pending for approval.', 1;

        --IF @RequireMakerResp = 1
        --   AND NOT EXISTS (SELECT 1
        --                   FROM   access.userAccesses AS ua
        --                          INNER JOIN
        --                          access.responsibilities AS r
        --                          ON ua.resp_syscode = r.resp_syscode
        --                   WHERE  ua.user_syscode = @UserSyscode
        --                          AND ua.is_active = 1
        --                          AND r.resp_name = 'Maker')
        --    THROW 71002, 'User lacks Maker responsibility.', 1;

        if @UserEntityCompCode is null or @UserEntityCompCode =0
        begin
         set @UserEntityCompCode=1;
        end

        IF NOT EXISTS (SELECT 1
                       FROM   access.users AS u
					   inner join access.userIdustryMapping uim on u.user_syscode=uim.user_syscode							 
                       WHERE  u.user_syscode = @UserSyscode
                              AND uim.indsCovGrp_code = @IndusCGCode
                              AND u.is_active = 1 and uim.is_active = 1)
            THROW 71003, 'User not authorized for industry coverage group.', 1;

        IF @EnforceFYOpen = 1
           AND NOT EXISTS (SELECT 1
                           FROM   masters.FinancialYears
                           WHERE  FY_Code = @FY_Code
                                  AND is_active = 1
                                  AND Status = 2)--'Open'
            THROW 71004, 'Financial year is not open for edits.', 1;
		
        IF @QuarterCode IS NULL
            BEGIN
                SELECT   TOP 1 @QuarterCode = quarter_code
                FROM     masters.Quarters
                WHERE    FY_Code = @FY_Code
                         AND Status IN (2,3)
                         AND is_active = 1
                ORDER BY quarter_start_date;
            END
        IF NOT EXISTS (SELECT 1
                       FROM   masters.Quarters
                       WHERE  quarter_code = @QuarterCode
                              AND FY_Code = @FY_Code
                              AND is_active = 1)
            THROW 71005, 'Quarter is invalid or not part of the financial year.', 1;

        DECLARE @QuarterStatus AS NVARCHAR (20) 
		--= (SELECT PolicyValue
  --                                                 FROM   masters.PolicyFlags
  --                                                 WHERE  PolicyCode = 'EnforceQuarterStatusForController');
        SELECT @QuarterStatus = Status
        FROM   masters.Quarters
        WHERE  quarter_code = @QuarterCode and FY_Code=@FY_Code;
        IF EXISTS (SELECT 1
                   FROM   access.roles
                   WHERE  role_code = @RoleCode
                          AND role_name = 'BudgetUser')
            BEGIN
                IF @EnforceQtrBudgetUser = 1
                   AND @QuarterStatus <> 2--'Open'
                    THROW 71006, 'Budget users can only add entries in Open quarters.', 1;
            END
        ELSE
            BEGIN
                IF @EnforceQtrController = 1
                   AND @QuarterStatus NOT IN (2, 3)--'Open', 'Frozen'
                    THROW 71007, 'Controllers can only add entries in Open or Frozen quarters.', 1;
            END
        IF EXISTS (SELECT 1
                   FROM   access.roles
                   WHERE  role_code = @RoleCode
                          AND role_name = 'Controller')
            BEGIN
                IF (@BET_Code IS NULL)
                   OR (@BET_Code NOT IN (3)
                       AND NOT ((@BET_Code = 1
                                 AND @AllowControllerAddBudget = 1)
                                OR (@BET_Code = 2
                                    AND @AllowControllerAddEstimated = 1)))
                    THROW 71008, 'Controllers can only add Actuals (unless policy allows Budget/Estimated).', 1;
                IF @BET_Code = 3
                   AND @BudgetBaseline_BEntrySys IS NULL
                    THROW 71009, 'Actuals require BudgetBaseline_BEntrySys referencing a baseline entry.', 1;
            END
        ELSE
            BEGIN
				if(@BET_Code is null)
				begin
					DECLARE @FirstQuarter AS INT 

					SELECT   TOP 1 @FirstQuarter = quarter_code
					FROM     masters.Quarters
					WHERE    FY_Code = @FY_Code
					ORDER BY quarter_start_date;
			
					SET @BET_Code = CASE WHEN @QuarterCode = @FirstQuarter THEN 1 ELSE 2 END;
				end
            END

        IF (
		--ISNULL(@Cproject_Code, 0) = 0
  --          AND ISNULL(@Ccompany_Code, 0) = 0  AND
			ISNULL(@Cproject_Name, '') = ''
            AND ISNULL(@Ccompany_Name, '') = '')
            THROW 71010, 'Either project or company must be provided (code or name).', 1;
        IF 
		--@Cindustry_code IS NULL AND
		ISNULL(@Cindustry_Name, '') <> ''
            BEGIN
                SELECT @Cindustry_code = Cindustry_code
                FROM   masters.ClientIndustry
                WHERE  Cindustry_name = @Cindustry_Name;
                IF @Cindustry_code IS NULL
                    BEGIN
					set @Cindustry_code=isnull((select max(Cindustry_code) from masters.ClientIndustry),0)+1;
                        INSERT  INTO masters.ClientIndustry (Cindustry_code,Cindustry_name, created_by)
                        VALUES  ( @Cindustry_code,@Cindustry_Name, @UserSyscode);
                        --SET @Cindustry_code = SCOPE_IDENTITY();
                    END
            END
			
        IF 	
		--@Ccompany_Code IS NULL AND 
		   ISNULL(@Ccompany_Name, '') <> ''
            BEGIN
                SELECT @Ccompany_Code = Ccompany_Code
                FROM   masters.ClientCompany
                WHERE  Ccompany_Name = @Ccompany_Name;
                IF @Ccompany_Code IS NULL
                    BEGIN
					set @Ccompany_Code=isnull((select max(Ccompany_Code) from masters.ClientCompany),0)+1
                        INSERT  INTO masters.ClientCompany (Ccompany_Code,Ccompany_Name, created_by, Cindustry_code)
                        VALUES                            (
						@Ccompany_Code,@Ccompany_Name, @UserSyscode, @Cindustry_code);
                        --SET @Ccompany_Code = SCOPE_IDENTITY();
                    END
            END
        IF 
		--@Cproject_Code IS NULL
  --         AND
		   ISNULL(@Cproject_Name, '') <> ''
            BEGIN
                SELECT @Cproject_Code = Cproject_Code
                FROM   masters.ClientProjects
                WHERE  Cproject_Name = @Cproject_Name;
                IF @Cproject_Code IS NULL
                    BEGIN
					set @Cproject_Code=isnull((select max(Cproject_Code) from masters.ClientProjects),0)+1
                        INSERT INTO masters.ClientProjects (Cproject_Code,Cproject_Name, created_by, Ccompany_Code)
                        VALUES (@Cproject_Code,@Cproject_Name, 
						@UserSyscode, @Ccompany_Code);
                        --SET @Cproject_Code = SCOPE_IDENTITY();
                    END
            END

			if @budgetEntry_Syscode >0
			begin
			    declare @GroupSys bigint

				SELECT @GroupSys = e.BudgetEntryGroup_Syscode
				FROM   BudgetEntry AS e  INNER JOIN  BudgetEntryGroup AS g
					   ON e.BudgetEntryGroup_Syscode = g.BudgetEntryGroup_Syscode
				WHERE  e.BudgetEntry_Syscode = @budgetEntry_Syscode;

			  if 	( select COUNT(be.BudgetEntry_Syscode) from BudgetEntry be inner join BudgetEntryGroup beg 
				on be.BudgetEntryGroup_Syscode =beg.BudgetEntryGroup_Syscode
				where be.BudgetEntryGroup_Syscode=@GroupSys and beg.Is_Active=1 and beg.Is_Deleted=0
				and be.Is_Deleted=0 and be.Status=0) > 0
					THROW 71010, 'Pending request is available for this budget.', 1;
			end

        DECLARE @BudgetEntryGroup_Syscode AS INT; 

        if(ISNULL( @budgetEGMaster_syscode,0)=0 )
        begin
            insert into BudgetEntryGroupMaster(is_active,created_date)
            values(1,getdate())
            SET @budgetEGMaster_syscode = SCOPE_IDENTITY();
        end

        INSERT  INTO BudgetEntryGroup (FY_Code, UserEntityCompCode, IndusCGCode, Cproject_Code, Ccompany_Code, Cindustry_code, BET_Code, Maker_Syscode,BudgetEGMaster_syscode)
        VALUES                       (@FY_Code, @UserEntityCompCode, @IndusCGCode, @Cproject_Code, @Ccompany_Code, @Cindustry_code, @BET_Code, @UserSyscode,@budgetEGMaster_syscode);
        SET @BudgetEntryGroup_Syscode = SCOPE_IDENTITY();
        DECLARE @NewEntrySys AS BIGINT 

        INSERT  INTO BudgetEntry (quarter_code, BudgetEntryGroup_Syscode,Supersedes_BEntrySys, BudgetBaseline_BEntrySys, 
		Product_code, ProdIssueType_code, IndicativeDealSize, EstimatedFeePool, TotalFees, NoOfBRLMs, JMFKeepRate, JMFGrossFee, 
		DealProbability, JMProbability, ExpectedClosure, Q1Fees, Q2Fees, Q3Fees, Q4Fees, TotalEstimate, Status, Maker_Syscode, 
		Created_On,IsMandated,net_amount,fee_amount,remarks,gross_amount)
        VALUES                  (@QuarterCode, @BudgetEntryGroup_Syscode,null, @BudgetBaseline_BEntrySys,  @Product_code,
		@ProdIssueType_code, @IndicativeDealSize, @EstimatedFeePool, @TotalFees, @NoOfBRLMs, @JMFKeepRate, @JMFGrossFee,
		@DealProbability, @JMProbability, @ExpectedClosure, @Q1Fees, @Q2Fees, @Q3Fees, @Q4Fees, @TotalEstimate, 0,
		@UserSyscode, GETDATE(),@isMandated,0,@fee_amount,@remarks,0);

        SET @NewEntrySys = SCOPE_IDENTITY();
          
            -- If no rows inserted, repopulate from master tables
            IF (SELECT COUNT(*) FROM #temp_budget_fee_sharing) = 0 and @BET_Code in (1,2)
            BEGIN
                TRUNCATE TABLE #temp_budget_fee_sharing;

                INSERT INTO #temp_budget_fee_sharing (
                    fee_type_code,
                    fee_type_name,
                    sharing_amt,
                    jmfs_amt,
                    jmfied_amt,
                    jmflibd_amt,
                    is_sharing_on
                )
                   SELECT distinct
                        f.fee_type_code,
                        f.fee_type_long_name as fee_type_name,
                        0 AS sharing_amt,
                        0 AS jmfs_amt,
                        0 AS jmfied_amt,
                        0 AS jmflibd_amt,
                        0 as is_sharing_on 
                    FROM fee_sharing_details fs  
                    left outer join masters.fee_type_master f on fs.fee_type_code =f.fee_type_code    
                    where f.is_active=1 and fs.is_active=1 and fs.bet_code=3 
                    and fs.fy_code=@FY_Code  order by f.fee_type_code;

            INSERT INTO budget_fee_details
                       (
                            BudgetEntry_Syscode,fee_code,Q1fee,Q2fee,Q3fee,Q4fee ,is_active ,created_by ,created_on ,modified_by ,modified_on
                       )
            select @NewEntrySys,fee_code,Q1fee,Q2fee,Q3fee,Q4fee,1,@UserSyscode,getdate(),null,null from #temp_fee_dtls
      	 
   
			select be.quarter_code into #actual_tbl from BudgetEntry be
			inner join BudgetEntryGroup beg on be.BudgetEntryGroup_Syscode=beg.BudgetEntryGroup_Syscode
			--inner join budget_fee_sharing_dtls fee on fee.BudgetEntry_Syscode=be.BudgetEntry_Syscode
			where be.BudgetEntryGroup_Syscode in (
			select BudgetEntryGroup_Syscode
			from BudgetEntryGroup
			where Status=1 --and (BudgetBaseline_BEntrySys = null
			and (be.BudgetBaseline_BEntrySys = (select top 1 BudgetBaseline_BEntrySys from BudgetEntry where BudgetEntry_Syscode = @BudgetEntry_Syscode)
			  or be.BudgetEntry_Syscode = (select top 1 BudgetBaseline_BEntrySys from BudgetEntry where BudgetEntry_Syscode = @BudgetEntry_Syscode))
			) and beg.BET_Code=3


            UPDATE tbs
            SET 
                tbs.sharing_amt =tfd.quaterfee,
                tbs.jmfs_amt =case when fsd.is_sharing_on =1 then  (per.jmfs_percentage / 100.0) * tfd.quaterfee else 0 end,
                tbs.jmfied_amt =case when fsd.is_sharing_on =1 then  (per.jmfied_percentage / 100.0) * tfd.quaterfee else 0 end,
                tbs.jmflibd_amt = case when fsd.is_sharing_on =1 
                then tfd.quaterfee - (((per.jmfs_percentage / 100.0) * tfd.quaterfee) + ((per.jmfied_percentage / 100.0) * tfd.quaterfee)) else tfd.quaterfee end
            FROM #temp_budget_fee_sharing tbs
                INNER JOIN (select  BudgetEntry_Syscode,fee_code,fee_type,

                  (
				   (CASE WHEN EXISTS (SELECT 1 FROM #actual_tbl WHERE quarter_code = 1) THEN 0 ELSE ISNULL(Q1fee,0) END) +
					(CASE WHEN EXISTS (SELECT 1 FROM #actual_tbl WHERE quarter_code = 2) THEN 0 ELSE ISNULL(Q2fee,0) END) +
					(CASE WHEN EXISTS (SELECT 1 FROM #actual_tbl WHERE quarter_code = 3) THEN 0 ELSE ISNULL(Q3fee,0) END) +
					(CASE WHEN EXISTS (SELECT 1 FROM #actual_tbl WHERE quarter_code = 4) THEN 0 ELSE ISNULL(Q4fee,0) END) 
				  --ISNULL(Q1fee,0) + ISNULL(Q2fee,0) + ISNULL(Q3fee,0) + ISNULL(Q4fee,0)
				  ) as quaterfee from #temp_fee_dtls) tfd
                    ON tbs.fee_type_code = tfd.fee_code
                inner join BudgetEntry b on b.BudgetEntry_Syscode=@NewEntrySys
                inner join BudgetEntryGroup bg on b.BudgetEntryGroup_Syscode=bg.BudgetEntryGroup_Syscode
                left join fee_sharing_details fsd on fsd.fee_type_code=tbs.fee_type_code and fsd.bet_code=@BET_Code
                left join masters.fee_percentage_details per on per.fy_code=bg.FY_Code 
                and per.bet_code=bg.BET_Code and per.issue_type_code=b.ProdIssueType_code
              
        end
        else
        begin
            if(@budgetEntry_Syscode >0)
            begin
                INSERT INTO budget_fee_details
                           (
                                BudgetEntry_Syscode,fee_code,Q1fee,Q2fee,Q3fee,Q4fee ,is_active ,created_by ,created_on ,modified_by ,modified_on
                           )
                select @NewEntrySys,fee_code,Q1fee,Q2fee,Q3fee,Q4fee,1,@UserSyscode,getdate(),null,null
                from budget_fee_details  where BudgetEntry_Syscode=@budgetEntry_Syscode
            end
   
        end

        --if(@BET_Code =3)
        --begin
             INSERT INTO budget_fee_sharing_dtls
                       ( BudgetEntry_Syscode,fee_type_code,sharing_amt,jmfs_amt,jmfied_amt,jmflibd_amt,is_active,created_by,created_on,modified_by,modified_on
                       )
            select @NewEntrySys,fee_type_code,sharing_amt,jmfs_amt,jmfied_amt,jmflibd_amt,1,@UserSyscode,getdate(),null,null from #temp_budget_fee_sharing
            
            set @net_amount=(select sum(jmflibd_amt) from budget_fee_sharing_dtls where BudgetEntry_Syscode=@NewEntrySys)
            set @gross_amount=(select  sum(sharing_amt) from budget_fee_sharing_dtls where BudgetEntry_Syscode=@NewEntrySys)

            if(@BET_Code =3)
            begin
               --select @Q1Fees=0, @Q2Fees=0,@Q3Fees=0,@Q4Fees=0
               declare @final_exp_closure_amt decimal(18,2)
                if(@QuarterCode=1)  
                begin                   
                    set @Q1Fees = @gross_amount;
                    set @final_exp_closure_amt= @fee_amount - @Q1Fees

                    if(@ExpectedClosure=1)
                        set @Q1Fees=@Q1Fees
                    else if (@ExpectedClosure=2)
                        set @Q2Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=3)
                        set @Q3Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=4)
                        set @Q4Fees = @final_exp_closure_amt
                end
                else if (@QuarterCode=2)
                begin
                  set @Q2Fees=@gross_amount;
                  set @final_exp_closure_amt= @fee_amount - @Q2Fees

                    if(@ExpectedClosure=1)
                        set @Q1Fees=@final_exp_closure_amt
                    else if (@ExpectedClosure=2)
                        set @Q2Fees =@Q2Fees
                    else if (@ExpectedClosure=3)
                        set @Q3Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=4)
                        set @Q4Fees = @final_exp_closure_amt
                end
                else if (@QuarterCode=3)
                begin
                  set @Q3Fees=@gross_amount;
                     set @final_exp_closure_amt= @fee_amount - @Q3Fees

                    if(@ExpectedClosure=1)
                        set @Q1Fees=@final_exp_closure_amt
                    else if (@ExpectedClosure=2)
                        set @Q2Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=3)
                        set @Q3Fees =@Q3Fees
                    else if (@ExpectedClosure=4)
                        set @Q4Fees = @final_exp_closure_amt
                end
                else if (@QuarterCode=4)
                begin
                  set @Q4Fees=@gross_amount;

                    set @final_exp_closure_amt= @fee_amount - @Q4Fees

                    if(@ExpectedClosure=1)
                        set @Q1Fees=@final_exp_closure_amt
                    else if (@ExpectedClosure=2)
                        set @Q2Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=3)
                        set @Q3Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=4)
                        set @Q4Fees = @Q4Fees
                end

                  update BudgetEntry
                        set Q1Fees=@Q1Fees,Q2Fees=@Q2Fees,Q3Fees=@Q3Fees,Q4Fees=@Q4Fees
                  where BudgetEntry_Syscode=@NewEntrySys;
            end

            update BudgetEntry set net_amount=@net_amount,gross_amount=@gross_amount where BudgetEntry_Syscode=@NewEntrySys;
       
       --end

        DECLARE @AuditEntityID AS INT = (SELECT AuditEntityID
                                         FROM   audits.AuditEntity
                                         WHERE  EntityName = 'BudgetEntry'), @AuditActionID AS INT = (SELECT AuditActionID
                                                                                                      FROM   audits.AuditAction
                                                                                                      WHERE  ActionName = 'Create');
        INSERT INTO audits.AuditTrail (AuditEntityID, EntityID, AuditActionID, ActionResult, ActorID, ActorResponsibility, Audit_On, ParentAuditID, Remarks)
        SELECT   @AuditEntityID,
                 @NewEntrySys,
                 @AuditActionID,
                 0,
                 @UserSyscode,
                 ua.resp_syscode,
                 GETDATE(),
                 NULL,
                 'Created via Add SP'
        FROM     access.userAccesses AS ua
        WHERE    ua.user_syscode = @UserSyscode
                 AND ua.is_active = 1
        ORDER BY ua.userAccess_Syscode
        OFFSET 0 ROWS FETCH NEXT 1 ROWS ONLY;
        COMMIT TRANSACTION;
        SET @ExecStatus = 1;
        SET @ResultMessage = 'Budget entry created successfully.';
		
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK;
        INSERT  INTO audits.ErrorLog (ProcName, Payload, ErrorMessage, ErrorNumber, ErrorSeverity, ErrorState, ErrorLine)
        VALUES                      ('sp_BudgetEntry_Add_JSON', @Payload, ERROR_MESSAGE(), ERROR_NUMBER(), ERROR_SEVERITY(), ERROR_STATE(), ERROR_LINE());
        SET @ExecStatus = 0;
        SET @ResultMessage = CONCAT('Failed: ', ERROR_MESSAGE());
		
        --THROW;
    END CATCH
END

DECLARE @AllowControllerAddBudget AS BIT = (SELECT PolicyValue

                       FROM   access.roles

            throw  71001, 'Change request already pending for approval.', 1;

           AND NOT EXISTS (SELECT 1
                           FROM   access.users AS u

                                  access.userRespMapping AS ur
                                  ON u.user_syscode = ur.user_syscode
								  INNER JOIN
                                  access.responsibilities AS r

                           WHERE  u.user_syscode = @UserSyscode
                                  AND u.is_active = 1 and ur.is_active=1
                                  AND r.resp_name = 'Maker')
            THROW 72002, 'User lacks Maker responsibility for editing.', 1;
        IF NOT EXISTS (SELECT 1
                       FROM   BudgetEntry
                       WHERE  BudgetEntry_Syscode = @TargetEntrySys
                              AND Is_Deleted = 0)
            THROW 72003, 'Target entry not found or deleted.', 1;

			DECLARE @EntryFY AS INT , @CurrentQuarter AS INT , @QuarterStatus AS NVARCHAR (20) , @EntryBET AS INT , 
			@GroupSys AS INT ;

        SELECT @GroupSys = e.BudgetEntryGroup_Syscode,
               @CurrentQuarter = e.quarter_code,
               @EntryBET = g.BET_Code,
               @EntryFY = g.FY_Code
        FROM   BudgetEntry AS e
               INNER JOIN
               BudgetEntryGroup AS g
               ON e.BudgetEntryGroup_Syscode = g.BudgetEntryGroup_Syscode
        WHERE  e.BudgetEntry_Syscode = @TargetEntrySys;

        IF @EnforceFYOpen = 1
           AND NOT EXISTS (SELECT 1
                           FROM   masters.FinancialYears
                           WHERE  FY_Code = @EntryFY
                                  AND is_active = 1
                                  AND Status = 2)--open cycle
            THROW 72004, 'Financial year is not open for edits.', 1;
        SET @QuarterCode = ISNULL(@QuarterCode, @CurrentQuarter);
        IF NOT EXISTS (SELECT 1
                       FROM   masters.Quarters
                       WHERE  quarter_code = @QuarterCode
                              AND FY_Code = @EntryFY
                              AND is_active = 1)
            THROW 72005, 'Quarter is invalid or not part of the entry''s financial year.', 1;

        SELECT @QuarterStatus = Status
        FROM   masters.Quarters
        WHERE  quarter_code = @QuarterCode and FY_Code = @EntryFY;

        IF EXISTS (SELECT 1
                   FROM   access.roles
                   WHERE  role_code = @RoleCode
                          AND role_name = 'BudgetUser')
            BEGIN
                IF @QuarterStatus <> 2 --open @EnforceQtrBudgetUser = 1 AND 
                    THROW 72006, 'Budget users can only edit entries in Open quarters.', 1;
                IF @EntryBET NOT IN (1, 2) --budgeted and estimated
                    THROW 72007, 'Budget users can only edit Budgeted or Estimated entries.', 1;
            END
        ELSE
            BEGIN
                IF @EnforceQtrController = 1
                   AND @QuarterStatus NOT IN (2, 3)--'Open', 'Frozen'
                    THROW 72008, 'Controllers can only edit entries in Open or Frozen quarters.', 1;
                IF @EntryBET IN (1)
                   AND @AllowControllerAddBudget = 0
                    THROW 72009, 'Controllers cannot edit Budgeted entries (policy).', 1;
                IF @EntryBET IN (2)
                   AND @AllowControllerAddEstimated = 0
                    THROW 72010, 'Controllers cannot edit Estimated entries (policy).', 1;
            END
        --UPDATE BudgetEntry
        --SET    Status        = 3,
        --       Superseded_On = GETDATE()
        --WHERE  BudgetEntry_Syscode = @TargetEntrySys;

		IF (
			ISNULL(@Cproject_Name, '') = ''
            AND ISNULL(@Ccompany_Name, '') = '')
            THROW 71010, 'Either project or company must be provided.', 1;

		IF ISNULL(@Cindustry_Name, '') <> ''
            BEGIN
                SELECT @Cindustry_code = Cindustry_code
                FROM   masters.ClientIndustry
                WHERE  Cindustry_name = @Cindustry_Name;
                IF @Cindustry_code IS NULL
                    BEGIN
					set @Cindustry_code=isnull((select max(Cindustry_code) from masters.ClientIndustry),0)+1;
                        INSERT  INTO masters.ClientIndustry (Cindustry_code,Cindustry_name, created_by)
                        VALUES  ( @Cindustry_code,@Cindustry_Name, @UserSyscode);
                    END
            END

		IF ISNULL(@Ccompany_Name, '') <> ''
            BEGIN
                SELECT @Ccompany_Code = Ccompany_Code
                FROM   masters.ClientCompany
                WHERE  Ccompany_Name = @Ccompany_Name;
                IF @Ccompany_Code IS NULL
                    BEGIN
					set @Ccompany_Code=isnull((select max(Ccompany_Code) from masters.ClientCompany),0)+1
                        INSERT  INTO masters.ClientCompany (Ccompany_Code,Ccompany_Name, created_by, Cindustry_code)
                        VALUES                            (
						@Ccompany_Code,@Ccompany_Name, @UserSyscode, @Cindustry_code);

		   IF  ISNULL(@Cproject_Name, '') <> ''
            BEGIN
                SELECT @Cproject_Code = Cproject_Code
                FROM   masters.ClientProjects
                WHERE  Cproject_Name = @Cproject_Name;
                IF @Cproject_Code IS NULL
                    BEGIN
					set @Cproject_Code=isnull((select max(Cproject_Code) from masters.ClientProjects),0)+1
                        INSERT INTO masters.ClientProjects (Cproject_Code,Cproject_Name, created_by, Ccompany_Code)
                        VALUES (@Cproject_Code,@Cproject_Name, 
						@UserSyscode, @Ccompany_Code);

        DECLARE @NewEntrySys AS BIGINT

		if @GroupSys >0
			begin

				where be.BudgetEntryGroup_Syscode=@GroupSys and beg.Is_Active=1 and beg.Is_Deleted=0

					THROW 71010, 'Pending request is available for this budget.', 1;

        INSERT INTO BudgetEntry (quarter_code, BudgetEntryGroup_Syscode, Supersedes_BEntrySys, BudgetBaseline_BEntrySys,

         IF (SELECT COUNT(*) FROM #temp_budget_fee_sharing) = 0 and @EntryBET in (1,2)

         if(@EntryBET in (1,2))
        begin
            INSERT INTO budget_fee_details

                       )
            select @NewEntrySys,fee_code,Q1fee,Q2fee,Q3fee,Q4fee,1,@UserSyscode,getdate(),null,null from #temp_fee_dtls

            UPDATE tbs
            SET 
                tbs.sharing_amt =tfd.quaterfee,
                tbs.jmfs_amt =case when tbs.is_sharing_on =1 then  (per.jmfs_percentage / 100.0) * tfd.quaterfee else 0 end,
                tbs.jmfied_amt =case when tbs.is_sharing_on =1 then  (per.jmfied_percentage / 100.0) * tfd.quaterfee else 0 end,
                tbs.jmflibd_amt = case when tbs.is_sharing_on =1 
                then tfd.quaterfee - (((per.jmfs_percentage / 100.0) * tfd.quaterfee) + ((per.jmfied_percentage / 100.0) * tfd.quaterfee)) else tfd.quaterfee end
            FROM #temp_budget_fee_sharing tbs
                INNER JOIN (select  BudgetEntry_Syscode,fee_code,fee_type,
                  (ISNULL(Q1fee,0) + ISNULL(Q2fee,0) + ISNULL(Q3fee,0) + ISNULL(Q4fee,0)) as quaterfee from #temp_fee_dtls) tfd
                    ON tbs.fee_type_code = tfd.fee_code
                inner join BudgetEntry b on b.BudgetEntry_Syscode=@NewEntrySys
                inner join BudgetEntryGroup bg on b.BudgetEntryGroup_Syscode=bg.BudgetEntryGroup_Syscode
                left join masters.fee_percentage_details per on per.fy_code=bg.FY_Code 
                and per.bet_code=bg.BET_Code and per.issue_type_code=b.ProdIssueType_code  
                left join fee_sharing_details fsd on fsd.fee_type_code=tbs.fee_type_code and fsd.bet_code=@EntryBET

            set @net_amount=(select sum(jmflibd_amt) from budget_fee_sharing_dtls where BudgetEntry_Syscode=@NewEntrySys)

                    else if (@ExpectedClosure=4)

                    else if (@ExpectedClosure=2)

                    else if (@ExpectedClosure=3)
                        set @Q3Fees =@final_exp_closure_amt
                    else if (@ExpectedClosure=4)

                     set @final_exp_closure_amt= @fee_amount - @Q3Fees

                    else if (@ExpectedClosure=2)

                    else if (@ExpectedClosure=3)

                    else if (@ExpectedClosure=4)

                    else if (@ExpectedClosure=2)
